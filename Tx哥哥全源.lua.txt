	local function fn4()
		if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("Humanoid") then
			return false
		end
		local humanoid = LocalPlayer.Character.Humanoid
		if str2 == "持续自转" then
			return true
		end

		if str2 == "移动时自转" then
			local humanoidRootPart = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
			if not humanoidRootPart then
				return false
			end
			local velocity = humanoidRootPart.Velocity
			return Vector3.new(velocity.X, 0, velocity.Z).Magnitude > 0.5
		end

		if str2 == "在地上时自转" then
			return humanoid.FloorMaterial ~= Enum.Material.Air
		end
		return false
	end

	local function fn5()
		if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
			return
		end
		local humanoidRootPart = LocalPlayer.Character.HumanoidRootPart
		fn3()

		if str == "BodyMover" then
			bodyAngularVelocity = Instance.new("BodyAngularVelocity")
			bodyAngularVelocity.MaxTorque = Vector3.new(0, math.huge, 0)
			bodyAngularVelocity.AngularVelocity = Vector3.new(0, n, 0)
			bodyAngularVelocity.Parent = humanoidRootPart

			connection = RunService_.Heartbeat:Connect(function()
				if not fn4() then
					bodyAngularVelocity.AngularVelocity = Vector3.zero
				else
					bodyAngularVelocity.AngularVelocity = Vector3.new(0, n, 0)
				end
			end)
		elseif str == "CFrame" then
			connection = RunService_.Heartbeat:Connect(function(deltaTime)
				if fn4() then
					humanoidRootPart.CFrame = humanoidRootPart.CFrame * CFrame.Angles(0, n * deltaTime, 0)
				end
			end)
		elseif str == "RotVelocity" then
			connection = RunService_.Heartbeat:Connect(function()
				if fn4() then
					humanoidRootPart.RotVelocity = Vector3.new(0, n, 0)
				else
					humanoidRootPart.RotVelocity = Vector3.zero
				end
			end)
		end
	end

	Default = false,
	Callback = function(value)
		if value then
			local currentCamera2 = Workspace.CurrentCamera
			local cFrame = currentCamera2.CFrame
			local character = LocalPlayer.Character
			if not character then
				return
			end
			local humanoid = character:FindFirstChildWhichIsA("Humanoid")
			if not humanoid then
				return
			end
			local clone = humanoid:Clone()
			clone.Parent = character
			LocalPlayer.Character = nil
			clone:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
			clone:SetStateEnabled(Enum.HumanoidStateType.GettingUp, false)
			clone.BreakJointsOnDeath = false
			humanoid:Destroy()
			LocalPlayer.Character = character
			currentCamera2.CameraSubject = clone
			currentCamera2.CFrame = cFrame
			clone.Health = clone.MaxHealth
			local animate = character:FindFirstChild("Animate")

			if animate then
				animate.Disabled = true
				task.wait()
				animate.Disabled = false
			end
		end
	end,
})

tbl.Tool:AddToggle("InvisibleMode", {
	Title = "隐身模式",
	Description = "部分服务器有效",
	Default = false,
	Callback = function(value)
		if value then
			local character = LocalPlayer.Character
			if not character then
				return
			end
			_G._originalChar = character
			character.Archivable = true
			_G._ghostChar = character:Clone()
			_G._ghostChar.Parent = Lighting
			_G._ghostChar.Name = ""
			local humanoidRootPart = character:FindFirstChild("HumanoidRootPart")

			if humanoidRootPart then
				_G._ghostChar:SetPrimaryPartCFrame(humanoidRootPart.CFrame)
			end

			character:MoveTo(Vector3.new(0, 999999, 0))
			character.Parent = Lighting
			_G._ghostChar.Parent = Workspace
			LocalPlayer.Character = _G._ghostChar
			local humanoid = _G._ghostChar:FindFirstChildWhichIsA("Humanoid")

			if humanoid then
				Workspace.CurrentCamera.CameraSubject = humanoid
			end
		elseif _G._originalChar then
			_G._originalChar.Parent = Workspace

			if _G._ghostChar then
				_G._ghostChar:Destroy()
				_G._ghostChar = nil
			end

			local humanoid = _G._originalChar:FindFirstChildWhichIsA("Humanoid")

			if humanoid then
				Workspace.CurrentCamera.CameraSubject = humanoid
			end

			_G._originalChar:MoveTo(Vector3.new(0, 10, 0))
			LocalPlayer.Character = _G._originalChar
			_G._originalChar = nil
		end
	end,
})
