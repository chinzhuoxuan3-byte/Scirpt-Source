local GeneralSection = GeneralTab:section("通用", true)
GeneralSection:Toggle("穿墙(需要演一下戏)", "text", false, function(state)
    local function enableNoclip(character)
        local humanoid = character:WaitForChild("Humanoid")
        local connection = RunService.Stepped:Connect(function()
            for _, part in pairs(character:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = false
                end
            end
        end)
        return connection
    end

    local noclipConnection

    if state then
        if LocalPlayer.Character then
            noclipConnection = enableNoclip(LocalPlayer.Character)
        end
        LocalPlayer.CharacterAdded:Connect(function(character)
            if noclipConnection then
                noclipConnection:Disconnect()
            end
            noclipConnection = enableNoclip(character)
        end)
    else
        if noclipConnection then
            noclipConnection:Disconnect()
        end
    end
end)

local ESPSCPTab = Window:Tab("透视SCP", "98541457845136")
local ESPSCPSection = ESPSCPTab:section("透视", true)

local function createESP(target, name)
    if not target then return end
    
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "ModelESP"
    billboard.Adornee = target
    billboard.Size = UDim2.new(0, 100, 0, 50)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true

    local label = Instance.new("TextLabel")
    label.Name = "TextLabel"
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.TextStrokeTransparency = 0
    label.TextSize = 14
    label.Font = Enum.Font.SourceSansBold
    label.Text = name .. "\n[SCP]"
    label.TextColor3 = Color3.new(0, 1, 0)
    label.Parent = billboard

    billboard.Parent = target

    -- 透视效果
    for _, part in pairs(target:GetDescendants()) do
        if part:IsA("BasePart") then
            part.LocalTransparencyModifier = 0
        end
    end
end
