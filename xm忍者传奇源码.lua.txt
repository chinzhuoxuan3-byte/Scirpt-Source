local autoCollectTask = nil
local coinPositions = {}
local currentCoinIndex = 0
local stayTime = 3
local switchDelay = 0.5

local function scanCoins()
    coinPositions = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local nameLower = string.lower(obj.Name)
            if string.find(nameLower, "hoop") then
                table.insert(coinPositions, obj.Position)
            end
        end
    end
    return coinPositions
end

CoinTab:Button({
    Title = "▶ 自动收集金币",
    Callback = function()
        scanCoins()
        if #coinPositions == 0 then
            WindUI:Notify({
                Duration = 2,
                Title = "错误",
                Content = "未找到金币圆环",
            })
            return
        end

        autoCollectTask = task.spawn(function()
            while true do
                for _, pos in ipairs(coinPositions) do
                    local character = LocalPlayer.Character
                    if character and character.Parent then
                        local hrp = character:FindFirstChild("HumanoidRootPart")
                        if hrp then
                            hrp.CFrame = CFrame.new(pos + Vector3.new(0, 5, 0))
                        end
                    end
                    task.wait(stayTime)
                    task.wait(1)
                    task.wait(switchDelay)
                end
            end
        end)

        WindUI:Notify({
            Duration = 2,
            Title = "启动",
            Content = "开始全图循环传送",
        })
    end,
})

CoinTab:Button({
    Title = " 停止循环传送",
    Callback = function()
        if autoCollectTask then
            task.cancel(autoCollectTask)
            autoCollectTask = nil
        end
        WindUI:Notify({
            Duration = 2,
            Title = "停止",
            Content = "循环传送已停止",
        })
    end,
})

CoinTab:Slider({
    Value = {
        Min = 1,
        Default = 3,

local ChiTab = Window:Tab({
    Locked = false,
    Title = " 自动集气",
    Icon = "wind",
})

local autoChiTask = nil
local chiPositions = {}
local chiWaitTime = 0.3

local function scanChi()
    chiPositions = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local nameLower = string.lower(obj.Name)
            if string.find(nameLower, "chi") and obj:FindFirstChild("TouchInterest") then
                table.insert(chiPositions, obj.Position)
            end
        end
    end
    return chiPositions
end

ChiTab:Button({
    Title = "▶ 自动收集气",
    Callback = function()
        scanChi()
        if #chiPositions == 0 then
            WindUI:Notify({
                Duration = 2,
                Title = "错误",
                Content = "未找到Chi",
            })
            return
        end

        autoChiTask = task.spawn(function()
            while true do
                for _, pos in ipairs(chiPositions) do
                    local character = LocalPlayer.Character
                    if character and character.Parent then
                        local hrp = character:FindFirstChild("HumanoidRootPart")
                        if hrp then
                            hrp.CFrame = CFrame.new(pos)
                        end
                    end
                    task.wait(chiWaitTime)
                end
            end
        end)

        WindUI:Notify({
            Duration = 2,
            Title = "启动",
            Content = "开始自动收集Chi",

})

local autoSwingTask = nil
local swingInterval = 0.01

TrainTab:Button({
    Title = "▶ 自动锻炼 (swingKatana)",
    Callback = function()
        local ninjaEvent = LocalPlayer:WaitForChild("ninjaEvent", 3)
        if not ninjaEvent then
            WindUI:Notify({
                Duration = 2,
                Title = "错误",
                Content = "未找到ninjaEvent",
            })
            return
        end

        autoSwingTask = task.spawn(function()
            while true do
                ninjaEvent:FireServer("swingKatana")
                task.wait(swingInterval)
            end
        end)

        WindUI:Notify({
            Duration = 2,
            Title = "启动",
            Content = "自动锻炼已开启",
        })
    end,
})

TrainTab:Button({
    Title = "  停止锻炼",
    Callback = function()
        if autoSwingTask then
            task.cancel(autoSwingTask)
            autoSwingTask = nil
        end
    end,
})

    Value = {
        Min = 0.005,

        swingInterval = value

})

    Buttons = {},

    Image = "",
    Title = "• 启用后，每次按下空格（跳跃）会自动向前冲刺设定的距离。\n• 冲刺期间有向上辅助力，跳跃更远更流畅。\n• 距离越大，前冲速度越快，请根据服务器容忍度调整。",
    Buttons = {},
    Color = "White",
    ImageSize = 0,
})

game:GetService("UserInputService").JumpRequest:Connect(function()
    if jumpEnabled then
        local character = LocalPlayer.Character
        if character and character.Parent then
            local hrp = character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local direction = hrp.CFrame.LookVector
                local targetPos = hrp.Position + direction * jumpDistance
                hrp.CFrame = CFrame.new(targetPos)
            end
        end
    end
end)
