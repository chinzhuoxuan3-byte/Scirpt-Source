local function updateCanCollide(updateCanCollideFlag)
	if not updateCanCollideFlag then
		return
	end

	for index, value in ipairs(updateCanCollideFlag:GetDescendants()) do
		if value:IsA("BasePart") then
			value.CanCollide = false
		end
	end
end

local function updateHeartbeatConnection()
	updateInstancePropertiesFlag = false

	if updateInstancePropertiesData.conns.headSit then
		updateInstancePropertiesData.conns.headSit:Disconnect()
		updateInstancePropertiesData.conns.headSit = nil
	end

	local updateHeartbeatConnectionCondition = getInstancePropertiesCharacter()

	if updateHeartbeatConnectionCondition then
		local humanoid = updateHeartbeatConnectionCondition:FindFirstChildOfClass("Humanoid")

		if humanoid then
			humanoid.Sit = false
		end

		secondaryUpdateCanCollide(updateHeartbeatConnectionCondition)
	end
end

local updateHeartbeatConnectionFlag = false

local function updateInstanceProperties(displayValue)
	if updateInstancePropertiesData.conns.headSit then
		updateInstancePropertiesData.conns.headSit:Disconnect()
		updateInstancePropertiesData.conns.headSit = nil
	end

	updateInstancePropertiesFlag = true
	local humanoidRootPartContainer = getInstancePropertiesCharacter()
	local humanoidRootPart

	if not humanoidRootPartContainer then
		updateInstancePropertiesFlag = false
		return
	else
		local humanoid = humanoidRootPartContainer:FindFirstChildOfClass("Humanoid")
		local flag = not humanoid
		humanoidRootPart = humanoidRootPartContainer:FindFirstChild("HumanoidRootPart")

		if flag or not humanoidRootPart then
			updateInstancePropertiesFlag = false
			return
		else
			updateCanCollide(humanoidRootPartContainer)
			humanoid.Sit = true
			local heartbeat = secondaryRunService.Heartbeat

			updateInstancePropertiesData.conns.headSit = heartbeat:Connect(function()
				pcall(function()
					if not updateInstancePropertiesFlag or not humanoidRootPartContainer.Parent then
						if updateHeartbeatConnectionFlag and localValuePlayer.Character then
							task.wait(1)

							if updateHeartbeatConnectionFlag and updateHeartbeatConnectionText ~= "" then
								updateInstanceProperties(updateHeartbeatConnectionText)
							end
						else
							updateHeartbeatConnection()
						end

						return
					else
						local secondaryPlayer = handleFlag(displayValue)

						if not secondaryPlayer or not secondaryPlayer.Character then
							if updateHeartbeatConnectionFlag then
								task.wait(1)

								if updateHeartbeatConnectionFlag and updateHeartbeatConnectionText ~= "" then
									updateInstanceProperties(updateHeartbeatConnectionText)
								end
							else
								updateHeartbeatConnection()
							end

							return
						else
							local secondaryHumanoidRootPart =
								secondaryPlayer.Character:FindFirstChild("HumanoidRootPart")

							if not secondaryHumanoidRootPart then
								return
							end

							humanoidRootPart.CFrame = secondaryHumanoidRootPart.CFrame * CFrame.new(0, 1.6, 0.4)
							updateCanCollide(humanoidRootPartContainer)
							return
						end
					end
				end)
			end)

			handleHeartbeatConnection("坐头启动", "已开始坐 " .. tostring(displayValue) .. " 的头")
			return
		end
	end
end

local function createEpixVel(item)
	local character = localValuePlayer.Character
	local cameraSubject = character and character:FindFirstChildOfClass("Humanoid")
	local rootPart = cameraSubject and cameraSubject.RootPart

	if not rootPart then
		return
	end

	local headContainer = item.Character
	local createEpixVelCondition, humanoid, updateEpixVel

	if not headContainer then
		return
	else
		humanoid = headContainer:FindFirstChildOfClass("Humanoid")
		createEpixVelCondition = humanoid and humanoid.RootPart
		local head = headContainer:FindFirstChild("Head")
		local accessory = headContainer:FindFirstChildOfClass("Accessory")
		local handle = accessory and accessory:FindFirstChild("Handle")

		if rootPart.Velocity.Magnitude < 50 then
			getgenv().OldPos = rootPart.CFrame
		end

		if humanoid and humanoid.Sit then
			return
		else
			if head then
				workspace.CurrentCamera.CameraSubject = head
			elseif handle then
				workspace.CurrentCamera.CameraSubject = handle
			elseif humanoid then
				workspace.CurrentCamera.CameraSubject = humanoid
			end

			function updateEpixVel(input, createEpixVelNumber, updateEpixVelNumber)
				rootPart.CFrame = CFrame.new(input.Position) * createEpixVelNumber * updateEpixVelNumber
				character:SetPrimaryPartCFrame(CFrame.new(input.Position) * createEpixVelNumber * updateEpixVelNumber)

				rootPart.Velocity = Vector3.new(90000000, 900000000, 90000000)
				rootPart.RotVelocity = Vector3.new(900000000, 900000000, 900000000)
			end

			local function handleEpixVel(input)
				local createEpixVelNumber = tick()
				local total = 0
				local sit

				repeat
					if rootPart and humanoid then
						if input.Velocity.Magnitude < 50 then
							total = total + 100

							updateEpixVel(
								input,
								CFrame.new(0, 1.5, 0) + humanoid.MoveDirection * input.Velocity.Magnitude / 1.25,
								CFrame.Angles(math.rad(total), 0, 0)
							)

							task.wait()

							updateEpixVel(
								input,
								CFrame.new(0, -1.5, 0) + humanoid.MoveDirection * input.Velocity.Magnitude / 1.25,
								CFrame.Angles(math.rad(total), 0, 0)
							)

							task.wait()

							updateEpixVel(
								input,
								CFrame.new(2.25, 1.5, -2.25) + humanoid.MoveDirection * input.Velocity.Magnitude / 1.25,
								CFrame.Angles(math.rad(total), 0, 0)
							)

							task.wait()

							updateEpixVel(
								input,
								CFrame.new(-2.25, -1.5, 2.25) + humanoid.MoveDirection * input.Velocity.Magnitude / 1.25,
								CFrame.Angles(math.rad(total), 0, 0)
							)

							task.wait()

							updateEpixVel(
								input,
								CFrame.new(0, 1.5, 0) + humanoid.MoveDirection,
								CFrame.Angles(math.rad(total), 0, 0)
							)

							task.wait()

							updateEpixVel(
								input,
								CFrame.new(0, -1.5, 0) + humanoid.MoveDirection,
								CFrame.Angles(math.rad(total), 0, 0)
							)

							task.wait()
						else
							updateEpixVel(
								input,
								CFrame.new(0, 1.5, humanoid.WalkSpeed),
								CFrame.Angles(math.rad(90), 0, 0)
							)

							task.wait()
							updateEpixVel(input, CFrame.new(0, -1.5, -humanoid.WalkSpeed), CFrame.Angles(0, 0, 0))
							task.wait()

							updateEpixVel(
								input,
								CFrame.new(0, 1.5, humanoid.WalkSpeed),
								CFrame.Angles(math.rad(90), 0, 0)
							)

							task.wait()

							updateEpixVel(
								input,
								CFrame.new(0, 1.5, createEpixVelCondition.Velocity.Magnitude / 1.25),
								CFrame.Angles(math.rad(90), 0, 0)
							)

							task.wait()

							updateEpixVel(
								input,
								CFrame.new(0, -1.5, -createEpixVelCondition.Velocity.Magnitude / 1.25),
								CFrame.Angles(0, 0, 0)
							)

							task.wait()

							updateEpixVel(
								input,
								CFrame.new(0, 1.5, createEpixVelCondition.Velocity.Magnitude / 1.25),
								CFrame.Angles(math.rad(90), 0, 0)
							)

							task.wait()
							updateEpixVel(input, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
							task.wait()
							updateEpixVel(input, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
							task.wait()
							updateEpixVel(input, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(-90), 0, 0))
							task.wait()
							updateEpixVel(input, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
							task.wait()
						end

						local isVelocityMagnitude = input.Velocity.Magnitude > 500

						sit = isVelocityMagnitude

						sit = isVelocityMagnitude
							or input.Parent ~= item.Character
							or item.Parent ~= players
							or not item.Character == headContainer
							or humanoid and humanoid.Sit
							or cameraSubject.Health <= 0
							or tick() > createEpixVelNumber + 2
					else
						break
					end
				until sit
			end

			workspace.FallenPartsDestroyHeight = (0 / 0)

			local epixVel = Instance.new("BodyVelocity")
			epixVel.Name = "EpixVel"
			epixVel.Parent = rootPart
			epixVel.Velocity = Vector3.new(900000000, 900000000, 900000000)
			epixVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)

			cameraSubject:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

			if createEpixVelCondition and head then
				if (createEpixVelCondition.CFrame.p - head.CFrame.p).Magnitude > 5 then
					handleEpixVel(head)
				else
					handleEpixVel(createEpixVelCondition)
				end
			elseif createEpixVelCondition then
				handleEpixVel(createEpixVelCondition)
			elseif head then
				handleEpixVel(head)
			elseif handle then
				handleEpixVel(handle)
			else
				return
			end

			epixVel:Destroy()
			cameraSubject:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
			workspace.CurrentCamera.CameraSubject = cameraSubject

			while true do
				rootPart.CFrame = getgenv().OldPos * CFrame.new(0, 0.5, 0)
				character:SetPrimaryPartCFrame(getgenv().OldPos * CFrame.new(0, 0.5, 0))
				cameraSubject:ChangeState("GettingUp")

				table.foreach(character:GetChildren(), function(createEpixVelArgument, secondaryArgument)
					if secondaryArgument:IsA("BasePart") then
						local vector = Vector3.new()
						local rotVelocity = Vector3.new()

						secondaryArgument.Velocity = vector
						secondaryArgument.RotVelocity = rotVelocity
					end
				end)

				task.wait()

				if (rootPart.Position - getgenv().OldPos.p).Magnitude < 25 then
					break
				end
			end

			workspace.FallenPartsDestroyHeight = getgenv().FPDH
			return
		end
	end
end

	end

		return

	if valueData.WalkSpeed.Enabled then
		humanoid.WalkSpeed = valueData.WalkSpeed.Value
	else
		humanoid.WalkSpeed = valueData.WalkSpeed.Default

local data = {}

local function fallbackHandler(argument, secondaryArgument)
	if data[argument] then
		data[argument]:Disconnect()
	end

	data[argument] = secondaryArgument
end

local updateCanCollideFlag = false
local updateCanCollideData = {}
local value10Data = {}

local function alternateUpdateCanCollide(updateCanCollideFlag)
	if not updateCanCollideFlag then
		return
	end

	for key, item in pairs(updateCanCollideFlag:GetDescendants()) do
		if item:IsA("BasePart") then
			if not value10Data[item] then
				value10Data[item] = item.CanCollide
			end

			item.CanCollide = false
		end
	end

	updateCanCollideData[updateCanCollideFlag] = updateCanCollideFlag.DescendantAdded:Connect(function(descendant)
		if descendant:IsA("BasePart") then
			if not value10Data[descendant] then
				value10Data[descendant] = descendant.CanCollide
			end

			descendant.CanCollide = false
		end
	end)
end

local connect, connection

local function additionalUpdateCanCollide()
	updateCanCollideFlag = false

	if connect then
		connect:Disconnect()
		connect = nil
	end

	if connection then
		connection:Disconnect()
		connection = nil
	end

	for key, canCollide in pairs(value10Data) do
		if key and key.Parent then
			key.CanCollide = canCollide
		end
	end

	for key, item in pairs(updateCanCollideData) do
		if item then
			item:Disconnect()
		end
	end

	updateCanCollideData = {}
	value10Data = {}
end

local function updateSteppedConnection()
	updateCanCollideFlag = true

	if connect then
		return
	else
		local character = secondaryLocalValuePlayer.Character

		if character then
			alternateUpdateCanCollide(character)
		end

		connect = runService.Stepped:Connect(function()
			if not updateCanCollideFlag then
				return
			else
				local character = secondaryLocalValuePlayer.Character

				if not character then
					return
				end

				for key, item in pairs(character:GetDescendants()) do
					if item:IsA("BasePart") then
						item.CanCollide = false
					end
				end

				return
			end
		end)

		if connection then
			connection:Disconnect()
		end

		connection = secondaryLocalValuePlayer.CharacterAdded:Connect(function(updateSteppedConnectionFlag)
			if updateCanCollideFlag then
				task.wait()
				alternateUpdateCanCollide(updateSteppedConnectionFlag)
			end
		end)

		return
	end
end

		end
	end

					end

					upData.moving.down = true
				end
			end)

					return

				elseif secondaryInput.KeyCode == Enum.KeyCode.S then
					upData.moving.backward = false
				elseif secondaryInput.KeyCode == Enum.KeyCode.A then

					upData.moving.down = false
				end
			end)

				if not upData.enabled then
					return

					local sum = Vector3.new(0, 0, 0)

					if upData.moving.forward then
						sum = sum + workspace.CurrentCamera.CFrame.LookVector
					end

					if upData.moving.backward then
						sum = sum - workspace.CurrentCamera.CFrame.LookVector
					end

					if upData.moving.left then
						sum = sum - workspace.CurrentCamera.CFrame.RightVector
					end

					if upData.moving.right then
						sum = sum + workspace.CurrentCamera.CFrame.RightVector
					end

					if upData.moving.up then
						sum = sum + Vector3.new(0, 1, 0)
					end

					if upData.moving.down then
						sum = sum + Vector3.new(0, -1, 0)
					end

					if sum.Magnitude > 0 then
						swimBodyVelocity.Velocity = sum.Unit * upData.speed
					else
						swimBodyVelocity.Velocity = Vector3.new(0, 0, 0)
					end

				end
			end)

					swimBodyVelocity.Velocity = Vector3.new(0, 0, 0)

					swimBodyGyro = Instance.new("BodyGyro")
					swimBodyGyro.Name = "SwimBodyGyro"
					swimBodyGyro.Parent = parent
					swimBodyGyro.MaxTorque = Vector3.new(2000, 2000, 2000)
					swimBodyGyro.D = 500
					swimBodyGyro.P = 8000
				end
			end)

			return
		end
	else
		local character = localValuePlayer.Character

		if character then
			local humanoid = character:FindFirstChildOfClass("Humanoid")

			if humanoid then
				for index, item in ipairs({
					Enum.HumanoidStateType.Climbing,
					Enum.HumanoidStateType.FallingDown,
					Enum.HumanoidStateType.Flying,
					Enum.HumanoidStateType.Freefall,
					Enum.HumanoidStateType.GettingUp,
					Enum.HumanoidStateType.Jumping,
					Enum.HumanoidStateType.Landed,
					Enum.HumanoidStateType.Physics,
					Enum.HumanoidStateType.PlatformStanding,
					Enum.HumanoidStateType.Ragdoll,
					Enum.HumanoidStateType.Running,
					Enum.HumanoidStateType.RunningNoPhysics,
					Enum.HumanoidStateType.Seated,
					Enum.HumanoidStateType.StrafingNoPhysics,
					Enum.HumanoidStateType.Swimming,
				}) do
					humanoid:SetStateEnabled(item, true)
				end

				humanoid:ChangeState(Enum.HumanoidStateType.RunningNoPhysics)
			end
		end

		for key, item in pairs(key19Data) do

				item:Disconnect()

			end

	if not humanoidRootPartContainer then
		return

		local humanoidRootPart = humanoidRootPartContainer:FindFirstChild("HumanoidRootPart")
		local humanoid = humanoidRootPartContainer:FindFirstChildOfClass("Humanoid")

		if humanoidRootPart and humanoid and humanoid.Health > 0 then
			if humanoidRootPart.Position.Y < self._voidY then
				if not self._platform then
					self:_createPlatform()
				end

				self._platform.Position =
					Vector3.new(humanoidRootPart.Position.X, self._voidY - 5, humanoidRootPart.Position.Z)

					CFrame.new(humanoidRootPart.Position.X, self._voidY + 3, humanoidRootPart.Position.Z)

				humanoidRootPart.Velocity = Vector3.new(0, 0, 0)

				self._platform:Destroy()
				self._platform = nil
			end

			self._platform:Destroy()
			self._platform = nil
		end

	end
end

		local rightLeg = rightLegContainer:FindFirstChild("Right Leg")

		local parent = rightLeg
		parent = rightLeg or rightLegContainer:FindFirstChild("RightLeg")

		if not parent then
			return
		else
			local specialMesh = parent:FindFirstChildOfClass("SpecialMesh")

			if specialMesh then
				specialMesh:Destroy()
			end

			local secondarySpecialMesh = Instance.new("SpecialMesh")
			secondarySpecialMesh.MeshId = "rbxassetid://6658418826"
			secondarySpecialMesh.TextureId = "rbxassetid://6658420819"
			secondarySpecialMesh.Scale = Vector3.new(1, 1, 1)
			secondarySpecialMesh.VertexColor = Vector3.new(1, 1, 1)
			secondarySpecialMesh.Parent = parent

			return
		end
	end
end
