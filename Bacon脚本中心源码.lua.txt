section3:Toggle("隐身(开启后需要重生才恢复)", "invisible", false, function(p8)
  getgenv().Invisible = p8
  local localPlayer9 = game.Players.LocalPlayer
  local v4 = {}

  local function f1(p9, p10)
    if not p9 then
      return
    end

    for index7, value7 in ipairs(p9:GetDescendants()) do
      if value7:IsA("BasePart") then
        if p10 == 1 and not v4[value7] then
          v4[value7] = value7.Transparency
        end

        value7.Transparency = p10
      end
    end

    for index8, value8 in ipairs(p9:GetDescendants()) do
      if value8:IsA("Decal") or value8:IsA("Texture") then
        if p10 == 1 and not v4[value8] then
          v4[value8] = value8.Transparency
        end

        value8.Transparency = p10
      end
    end

    for index9, value9 in ipairs(p9:GetDescendants()) do
      if value9:IsA("SpecialMesh") or value9:IsA("BlockMesh") or value9:IsA("CylinderMesh") then
        if p10 == 1 and not v4[value9] then
          v4[value9] = value9.Transparency
        end

        value9.Transparency = p10
      end
    end
  end

  local function f2(p11)
    if not p11 then
      return
    end

    for index10, value10 in ipairs(p11:GetDescendants()) do
      if value10:IsA("BasePart") and v4[value10] then
        value10.Transparency = v4[value10]
      end
    end

    for index11, value11 in ipairs(p11:GetDescendants()) do
      if (value11:IsA("Decal") or value11:IsA("Texture")) and v4[value11] then
        value11.Transparency = v4[value11]
      end
    end

    for index12, value12 in ipairs(p11:GetDescendants()) do
      if (value12:IsA("SpecialMesh") or value12:IsA("BlockMesh") or value12:IsA("CylinderMesh"))
        and v4[value12] then
        value12.Transparency = v4[value12]
      end
    end

    v4 = {}
  end

  local invisibleLoop

  if p8 then
    if localPlayer9.Character then
      f1(localPlayer9.Character, 1)
    end

    getgenv().InvisibleLoop = true
    invisibleLoop = getgenv().InvisibleLoop

    task.spawn(function()
      while getgenv().Invisible and getgenv().InvisibleLoop == invisibleLoop do
        task.wait(0.1)

        if localPlayer9.Character then
          f1(localPlayer9.Character, 1)
        end
      end
    end)

    if getgenv().InvisibleCharConnection then
      getgenv().InvisibleCharConnection:Disconnect()
    end

    local v5 = getgenv()

    v5.InvisibleCharConnection = localPlayer9.CharacterAdded:Connect(function(character9)
      task.wait(0.1)

      if getgenv().Invisible then
        f1(character9, 1)
      end
    end)
  else
    getgenv().InvisibleLoop = false

    if getgenv().InvisibleCharConnection then
      getgenv().InvisibleCharConnection:Disconnect()
      getgenv().InvisibleCharConnection = nil
    end

    if localPlayer9.Character then
      f2(localPlayer9.Character)
    end

    v4 = {}
  end
end)
